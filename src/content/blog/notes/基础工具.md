## 主要内容

## uv

```cmd
uv python list #查看可用python

uv python install 3.12 # 安装特定python

uv venv --python 3.12 .venv # 创建虚拟环境
uv venv --python 3.12
uv venv 

# macOS/Linux
source .venv/bin/activate
# Windows
.venv\Scripts\activate

deactivate

# 安装包，与pip命令替代
uv pip install requests
uv pip install requests=2.31.0
uv pip install -r requirements.txt

# 导出环境依赖
uv pip freeze > requirements.txt

# 项目管理 使用pyproject.toml,比pip更推荐团队协作
uv init my_project
cd my_project
# 可创建基本项目结构

# 添加依赖
uv add requests
uv add "requests>=2.31.0"
uv remove requests

# 克隆项目或更新pyproject.toml后安装所有依赖
uv sync

# 运行脚本
uv run main.py 无需手动激活虚拟环境
```
